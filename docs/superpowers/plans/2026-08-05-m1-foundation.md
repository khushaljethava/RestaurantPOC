# M1 Foundation Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Stand up the tenancy foundation — six stall owners can register, run a food court group, manage menus and staff, and no owner can reach another owner's private data.

**Architecture:** Supabase Postgres is the entire backend; there is no custom API layer. Authorization lives in Row Level Security, driven by two `SECURITY DEFINER` helper functions (`my_stall_ids()`, `my_group_stall_ids()`). A Next.js App Router admin talks to Supabase directly. Every RLS policy is tested with pgTAP before any UI is written, because RLS is the security boundary and a bug there leaks one owner's business to another.

**Tech Stack:** Postgres 15 (Supabase local), pgTAP, Supabase CLI 2.111 (via `npx`), Next.js 15 App Router, TypeScript, Tailwind, Supabase JS v2, Docker.

## Global Constraints

- **Spec:** `docs/superpowers/specs/2026-08-05-cross-stall-pos-design.md`. Where this plan and the spec disagree, the spec wins.
- **The stall is the tenant.** Never introduce a "business" or "branch" table above it.
- **`stall.group_id` is nullable.** A standalone stall must work with zero group rows and no special-case code.
- **Staff, invite codes, and stall settings are stall-private** — never readable by other group members, no exceptions.
- **Menus, orders, bills are group-readable** — any stall in the same group may read them.
- All helper functions are `SECURITY DEFINER STABLE` with `SET search_path = ''` and fully-qualified table names. RLS on `staff_membership` would otherwise recurse infinitely when a policy queries that same table.
- Every table with RLS gets a pgTAP test proving both the allow case and the deny case. A policy with only an allow test is untested.
- Supabase CLI is invoked as `npx supabase@2.111.0` — it is not installed globally.
- Money is `numeric(12,2)`. Never `float`.
- Commit after every task. Conventional commit prefixes (`feat:`, `test:`, `chore:`).

```sql
-- A food court is a federation of independent stall owners.
-- It owns no menus, no revenue and no staff.
create table public.food_court (
  id          uuid primary key default gen_random_uuid(),
  name        text not null check (length(trim(name)) > 0),
  created_by  uuid not null references auth.users (id),
  created_at  timestamptz not null default now()
);

-- THE TENANT. group_id is nullable: a standalone single store is simply
-- a stall with no group, not a special case in application code.
create table public.stall (
  id          uuid primary key default gen_random_uuid(),
  name        text not null check (length(trim(name)) > 0),
  group_id    uuid references public.food_court (id) on delete set null,
  gst_no      text,
  currency    text not null default 'INR',
  timezone    text not null default 'Asia/Kolkata',
  logo_url    text,
  is_open     boolean not null default true,
  created_by  uuid not null references auth.users (id),
  created_at  timestamptz not null default now()
);

create index stall_group_id_idx on public.stall (group_id);

create table public.staff_membership (
  id         uuid primary key default gen_random_uuid(),
  stall_id   uuid not null references public.stall (id) on delete cascade,
  user_id    uuid not null references auth.users (id) on delete cascade,
  role       text not null check (role in ('owner', 'manager', 'staff')),
  created_at timestamptz not null default now(),
  unique (stall_id, user_id)
);

create index staff_membership_user_id_idx on public.staff_membership (user_id);
```

```sql
-- SECURITY DEFINER is mandatory here, not a convenience.
-- RLS on staff_membership will itself call my_stall_ids(); if this function
-- ran with the caller's rights it would re-enter that policy forever.
-- Definer rights bypass RLS and break the cycle.
--
-- SET search_path = '' forces fully-qualified names so a caller cannot
-- shadow `public` with their own schema and hijack the lookup.

create or replace function public.my_stall_ids()
returns setof uuid
language sql
stable
security definer
set search_path = ''
as $$
  select m.stall_id
  from public.staff_membership m
  where m.user_id = (select auth.uid());
$$;

comment on function public.my_stall_ids() is
  'Stalls the caller is a member of. Basis for stall-private access.';

create or replace function public.my_group_stall_ids()
returns setof uuid
language sql
stable
security definer
set search_path = ''
as $$
  -- own stalls
  select s.id
  from public.stall s
  where s.id in (select public.my_stall_ids())
  union
  -- every stall sharing a group with one of my stalls.
  -- group_id is not null on both sides, so a standalone stall (null group)
  -- matches nothing here and falls back to the first branch alone.
  select peer.id
  from public.stall peer
  join public.stall mine
    on mine.group_id = peer.group_id
  where mine.id in (select public.my_stall_ids())
    and peer.group_id is not null;
$$;

comment on function public.my_group_stall_ids() is
  'Own stalls plus food-court peers. Basis for group-readable data.';

grant execute on function public.my_stall_ids() to authenticated;
grant execute on function public.my_group_stall_ids() to authenticated;
```

Create `supabase/migrations/0004_tenancy_rls.sql`:

```sql
alter table public.stall            enable row level security;
alter table public.staff_membership enable row level security;
alter table public.food_court       enable row level security;

-- STALL ---------------------------------------------------------------
-- Readable across the group: you must see a peer's stall to order from it.
create policy stall_group_member_can_read
  on public.stall for select
  to authenticated
  using (id in (select public.my_group_stall_ids()));

-- Writable only by an owner or manager OF THAT STALL.
create policy stall_owner_can_update
  on public.stall for update
  to authenticated
  using (
    exists (
      select 1 from public.staff_membership m
      where m.stall_id = stall.id
        and m.user_id = (select auth.uid())
        and m.role in ('owner', 'manager')
    )
  );

-- Anyone signed in may create a stall; they become its creator.
create policy stall_authenticated_can_insert
  on public.stall for insert
  to authenticated
  with check (created_by = (select auth.uid()));

-- STAFF MEMBERSHIP ----------------------------------------------------
-- Stall-private. Group membership grants NOTHING here: revenue sharing is a
-- business decision, but letting a peer add users to your stall is an attack.
create policy staff_own_stall_can_read
  on public.staff_membership for select
  to authenticated
  using (stall_id in (select public.my_stall_ids()));

create policy staff_owner_can_write
  on public.staff_membership for all
  to authenticated
  using (
    exists (
      select 1 from public.staff_membership m
      where m.stall_id = staff_membership.stall_id
        and m.user_id = (select auth.uid())
        and m.role = 'owner'
    )
  )
  with check (
    exists (
      select 1 from public.staff_membership m
      where m.stall_id = staff_membership.stall_id
        and m.user_id = (select auth.uid())
        and m.role = 'owner'
    )
  );

-- FOOD COURT ----------------------------------------------------------
create policy food_court_member_can_read
  on public.food_court for select
  to authenticated
  using (
    id in (
      select s.group_id from public.stall s
      where s.id in (select public.my_stall_ids())
        and s.group_id is not null
    )
  );

create policy food_court_authenticated_can_insert
  on public.food_court for insert
  to authenticated
  with check (created_by = (select auth.uid()));
```

Note the first membership row is a chicken-and-egg case: `staff_owner_can_write` requires an existing owner row. Task 5's `register_stall()` RPC runs as `SECURITY DEFINER` and creates it.

```sql
returns uuid
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_uid     uuid := auth.uid();
  v_group   public.group_invite;
  v_request uuid;

  if not exists (
    select 1 from public.staff_membership m
    where m.stall_id = p_stall_id and m.user_id = v_uid and m.role = 'owner'

    raise exception 'only an owner may join a group' using errcode = '42501';

  returning id into v_request;

end;

returns void

begin
  select * into v_req from public.group_join_request where id = p_request_id;
  if not found then
    raise exception 'no such request' using errcode = 'P0002';
  end if;

  if not exists (
    select 1 from public.stall s
    join public.staff_membership m on m.stall_id = s.id

      and m.user_id = v_uid
      and m.role = 'owner'

  update public.group_join_request set status = 'approved' where id = p_request_id;
end;
```

```typescript
import { createServerClient } from '@supabase/ssr';
import { NextResponse, type NextRequest } from 'next/server';

export async function middleware(request: NextRequest) {
  let response = NextResponse.next({ request });

  const supabase = createServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
    {
      cookies: {
        getAll: () => request.cookies.getAll(),
        setAll: (list) => {
          list.forEach(({ name, value }) => request.cookies.set(name, value));
          response = NextResponse.next({ request });
          list.forEach(({ name, value, options }) =>
            response.cookies.set(name, value, options)
          );
        },
      },
    }
  );

  const { data: { user } } = await supabase.auth.getUser();

  const isPublic = request.nextUrl.pathname.startsWith('/login');
  if (!user && !isPublic) {
    return NextResponse.redirect(new URL('/login', request.url));
  }
  return response;
}

export const config = {
  matcher: ['/((?!_next/static|_next/image|favicon.ico).*)'],
};
```

```typescript
import { revalidatePath } from 'next/cache';
import { createClient } from '@/lib/supabase/server';

export async function createItem(formData: FormData) {
  const supabase = await createClient();
  const { error } = await supabase.from('menu_item').insert({
    stall_id: String(formData.get('stall_id')),
    name: String(formData.get('name')),
    price: Number(formData.get('price')),
    tax_rate: Number(formData.get('tax_rate') || 0),
  });
  if (error) throw new Error(error.message);
  revalidatePath('/menu');
}

export async function toggleAvailability(id: string, next: boolean) {
  const supabase = await createClient();
  const { error } = await supabase
    .from('menu_item')
    .update({ is_available: next })
    .eq('id', id);
  if (error) throw new Error(error.message);
  revalidatePath('/menu');
}
```
